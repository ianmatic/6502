BALL_POS_L = $00 ; ball's current position
BALL_POS_H = $01
BALL_PREV_POS_L = $02 ; ball's previous position
BALL_PREV_POS_H = $03
BALL_COLOR_VAL = $01 ; White

PADDLE_OFFSET = $04 ; paddle drawing offset
PADDLE_OFFSET_COUNTER = $05 ; paddle offset counter
PADDLE_HEIGHT_VAL = $07 ; the height (number of pixels) of each paddle

P1_POSITION_TOP_L = $10
P1_POSITION_TOP_H = $11
P1_POSITION_BOTTOM_L = $12
P1_POSITION_BOTTOM_H = $13

P1_POSITION_TOP_PREV_L = $14
P1_POSITION_TOP_PREV_H = $15
P1_POSITION_BOTTOM_PREV_L = $16
P1_POSITION_BOTTOM_PREV_H = $17
P1_MOVEMENT_FLAG = $18 ; flag to indicate P1 movement

P2_POSITION_TOP_L = $20
P2_POSITION_TOP_H = $21
P2_MOVEMENT_FLAG = $22 ; flag to indicate P2 movement

MOVEMENT_NONE_VAL = $00
MOVEMENT_UP_VAL = $01
MOVEMENT_DOWN_VAL = $02

P1_COLOR_VAL = $0a
P2_COLOR_VAL = $0e

KEY_PRESS = $ff
W_KEY_VAL = $77
S_KEY_VAL = $73
O_KEY_VAL = $6f
L_KEY_VAL = $6c

ROW_SHIFT_VAL = $20 ; hex amount required to shift up or down 1 row

jsr setup
jmp loop

setup:
  jsr setupP1
  jsr setupP2
  jsr setupBall
  rts

setupP1:
  ; $03a2 is the top initial point for P1's position
  lda #$a2
  sta P1_POSITION_TOP_L
  lda #$03
  sta P1_POSITION_TOP_H

  ; $0462 is the bottom initial point for P1's position
  lda #$62
  sta P1_POSITION_BOTTOM_L
  lda #$04
  sta P1_POSITION_BOTTOM_H
  rts

setupP2:
  ; $03be is the starting top point for P1's position
  lda #$be
  sta P2_POSITION_TOP_L
  lda #$03
  sta P2_POSITION_TOP_H
  rts

setupBall:
  ; $0410 is the center of the screen
  lda #$10
  sta BALL_POS_L
  lda #$04
  sta BALL_POS_H
  rts

loop:
  jsr stall
  jsr handleInput
  jsr updateBall
  jsr updateP1
  jsr updateP2
  jsr draw
  jmp loop

stall:
  ldx #$ff
stallInner:
  nop
  nop
  dex
  bne stallInner
  rts

handleInput:
  lda KEY_PRESS
  jsr handleP1Input
  jsr handleP2Input
  rts

handleP1Input:
  CMP #W_KEY_VAL
  BNE checkSKey
  ldx #MOVEMENT_UP_VAL
  jmp returnP1Input
checkSKey:
  CMP #S_KEY_VAL
  BNE noP1Input
  ldx #MOVEMENT_DOWN_VAL
  jmp returnP1Input
noP1Input:
  ldx #MOVEMENT_NONE_VAL
returnP1Input:
  stx P1_MOVEMENT_FLAG
  rts

handleP2Input:
  CMP #O_KEY_VAL
  BNE checkLKey
  ldx #MOVEMENT_UP_VAL
  jmp returnP2Input
checkLKey:
  CMP #L_KEY_VAL
  BNE noP2Input
  ldx #MOVEMENT_DOWN_VAL
  jmp returnP2Input
noP2Input:
  ldx #MOVEMENT_NONE_VAL
returnP2Input:
  stx P2_MOVEMENT_FLAG
  rts

updateBall:
  ; Copy current position into prev position
  ldx BALL_POS_L
  stx BALL_PREV_POS_L
  ldx BALL_POS_H
  stx BALL_PREV_POS_H

  ; Move ball to the right
  ldx BALL_POS_L
  inx
  stx BALL_POS_L

  rts
updateP1:
  lda P1_POSITION_TOP_L
  sta P1_POSITION_TOP_PREV_L

  lda P1_POSITION_TOP_H
  sta P1_POSITION_TOP_PREV_H

  lda P1_POSITION_BOTTOM_L
  sta P1_POSITION_BOTTOM_PREV_L

  lda P1_POSITION_BOTTOM_H
  sta P1_POSITION_BOTTOM_PREV_H

  ldx P1_MOVEMENT_FLAG
  CPX #MOVEMENT_UP_VAL
  BEQ moveP1Up
  CPX #MOVEMENT_DOWN_VAL
  BEQ moveP1Down
  rts

moveP1Up:
  sec 
  lda P1_POSITION_TOP_L
  sbc #ROW_SHIFT_VAL
  sta P1_POSITION_TOP_L

  lda P1_POSITION_TOP_H
  sbc #$00
  sta P1_POSITION_TOP_H

  sec 
  lda P1_POSITION_BOTTOM_L
  sbc #ROW_SHIFT_VAL
  sta P1_POSITION_BOTTOM_L

  lda P1_POSITION_BOTTOM_H
  sbc #$00
  sta P1_POSITION_BOTTOM_H

  rts
moveP1Down:

  clc 
  lda P1_POSITION_TOP_L
  adc #ROW_SHIFT_VAL
  sta P1_POSITION_TOP_L

  lda P1_POSITION_TOP_H
  adc #$00
  sta P1_POSITION_TOP_H

  clc 
  lda P1_POSITION_BOTTOM_L
  adc #ROW_SHIFT_VAL
  sta P1_POSITION_BOTTOM_L

  lda P1_POSITION_BOTTOM_H
  adc #$00
  sta P1_POSITION_BOTTOM_H

  rts

updateP2:
  ; P2 controls: O (6f) & L (6c)
  rts

draw:
  jsr drawBall
  jsr drawP1
  jsr drawP2
  rts

drawBall:
  ldy #$00
  
  ; Erase previous position
  lda #$0
  sta (BALL_PREV_POS_L), Y

  ; Draw current position
  lda #BALL_COLOR_VAL
  sta (BALL_POS_L), Y
  rts

drawP1:
  ; erase pixels from movement
  lda P1_MOVEMENT_FLAG
  cmp #MOVEMENT_UP_VAL
  beq removeBottomP1Pixel
  cmp #MOVEMENT_DOWN_VAL
  beq removeTopP1Pixel

zeroOut:
  ; Zero out offset and offset counter
  lda #$00
  sta PADDLE_OFFSET
  sta PADDLE_OFFSET_COUNTER

  jsr drawP1Loop
  rts

removeBottomP1Pixel:
  lda #$00
  ldy #$00
  sta (P1_POSITION_BOTTOM_PREV_L), Y
  jmp zeroOut
removeTopP1Pixel:
  lda #$00
  ldy #$00
  sta (P1_POSITION_TOP_PREV_L), Y
  jmp zeroOut

drawP1Loop:
  ; Get latest offset counter
  ldx PADDLE_OFFSET_COUNTER
  inx
  stx PADDLE_OFFSET_COUNTER

  ; Draw current pixel
  ldy PADDLE_OFFSET
  lda #P1_COLOR_VAL
  sta (P1_POSITION_TOP_L), Y

  ; Update offset
  tya 
  clc
  adc #ROW_SHIFT_VAL
  sta PADDLE_OFFSET

  ; Update offset counter
  cpx #PADDLE_HEIGHT_VAL
  bne drawP1Loop
  rts

drawP2:

  ; Zero out offset and offset counter
  lda #$00
  sta PADDLE_OFFSET
  sta $05

  jsr drawP2Loop
  rts

drawP2Loop:
  ; Get latest offset counter
  ldx PADDLE_OFFSET_COUNTER
  inx
  stx PADDLE_OFFSET_COUNTER

  ; Draw current pixel
  ldy PADDLE_OFFSET
  lda #P2_COLOR_VAL
  sta (P2_POSITION_TOP_L), Y

  ; Update offset
  tya 
  clc
  adc #ROW_SHIFT_VAL
  sta PADDLE_OFFSET

  ; Update offset counter
  cpx #PADDLE_HEIGHT_VAL
  bne drawP2Loop
  rts

; Memory location $fe contains a new random byte on every instruction.
; Memory location $ff contains the ascii code of the last key pressed.
; 
; Memory locations $200 to $5ff map to the screen pixels. Different values will
; draw different colour pixels. The colours are:
; 
; $0: Black
; $1: White
; $2: Red
; $3: Cyan
; $4: Purple
; $5: Green
; $6: Blue
; $7: Yellow
; $8: Orange
; $9: Brown
; $a: Light red
; $b: Dark grey
; $c: Grey
; $d: Light green
; $e: Light blue
; $f: Light grey

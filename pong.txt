BALL_POS_L = $00 ; ball's current position
BALL_POS_H = $01
BALL_PREV_POS_L = $02 ; ball's previous position
BALL_PREV_POS_H = $03
BALL_COLOR_VAL = $01 ; White

PADDLE_OFFSET = $04 ; paddle drawing offset
PADDLE_OFFSET_COUNTER = $05 ; paddle offset counter
PADDLE_HEIGHT_VAL = $07 ; the height (number of pixels) of each paddle

P1_POSITION_L = $10
P1_POSITION_H = $11
P1_MOVEMENT_FLAG = $12 ; flag to indicate P1 movement

P2_POSITION_L = $20
P2_POSITION_H = $21
P2_MOVEMENT_FLAG = $22 ; flag to indicate P2 movement

MOVEMENT_NONE_VAL = $00
MOVEMENT_UP_VAL = $01
MOVEMENT_DOWN_VAL = $02

P1_COLOR_VAL = $0a
P2_COLOR_VAL = $0e

KEY_PRESS = $ff
W_KEY_VAL = $77
S_KEY_VAL = $73

ROW_SHIFT_VAL = $20 ; hex amount required to shift up or down 1 row

jsr setup
jmp loop

setup:
  jsr setupP1
  jsr setupP2
  jsr setupBall
  rts

setupP1:
  ; $03a2 is the starting point for P1's position
  lda #$a2
  sta $10
  lda #$03
  sta $11
  rts

setupP2:
  ; $03be is the starting point for P1's position
  lda #$be
  sta $20
  lda #$03
  sta $21
  rts

setupBall:
  ; $0410 is the center of the screen
  lda #$10
  sta BALL_POS_L
  lda #$04
  sta BALL_POS_H
  rts

loop:
  jsr stall
  jsr updateBall
  jsr updateP1
  jsr updateP2
  jsr draw
  jmp loop


stall:
  ldx #$ff
stallInner:
  nop
  nop
  dex
  bne stallInner
  rts

updateBall:
  ; Copy current position into prev position
  ldx BALL_POS_L
  stx BALL_PREV_POS_L
  ldx BALL_POS_H
  stx BALL_PREV_POS_H

  ; Move ball to the right
  ldx BALL_POS_L
  inx
  stx BALL_POS_L

  rts
updateP1:

  ; P1 controls
  lda KEY_PRESS
  CMP #W_KEY_VAL
  BEQ moveP1Up
  CMP #S_KEY_VAL
  BEQ moveP1Down

  ; No movement
  ldx #MOVEMENT_NONE_VAL
  stx P1_MOVEMENT_FLAG
  rts
moveP1Up:
  ldx #MOVEMENT_UP_VAL
  stx P1_MOVEMENT_FLAG
  jmp updateP2
moveP1Down:
  ldx #MOVEMENT_DOWN_VAL
  stx P1_MOVEMENT_FLAG
  jmp updateP2

updateP2:
  ; P2 controls: O (6f) & L (6c)
  rts

draw:
  jsr drawBall
  jsr drawP1
  jsr drawP2
  rts

drawBall:
  ldy #$00
  
  ; Erase previous position
  lda #$0
  sta (BALL_PREV_POS_L), Y

  ; Draw current position
  lda #BALL_COLOR_VAL
  sta (BALL_POS_L), Y
  rts

drawP1:
  ; erase pixels from movement
  ; lda P1_MOVEMENT_FLAG
  ; cmp #MOVEMENT_UP_VAL
  ; beq removeBottomP1Pixel
  ; cmp #MOVEMENT_DOWN_VAL
  ; beq removeTopP2Pixel

  ; Zero out offset and offset counter
  lda #$00
  sta PADDLE_OFFSET
  sta PADDLE_OFFSET_COUNTER

  jsr drawP1Loop
  rts

drawP1Loop:
  ; Get latest offset counter
  ldx PADDLE_OFFSET_COUNTER
  inx
  stx PADDLE_OFFSET_COUNTER

  ; Draw current pixel
  ldy PADDLE_OFFSET
  lda #P1_COLOR_VAL
  sta (P1_POSITION_L), Y

  ; Update offset
  tya 
  clc
  adc #ROW_SHIFT_VAL
  sta PADDLE_OFFSET

  ; Update offset counter
  cpx #PADDLE_HEIGHT_VAL
  bne drawP1Loop
  rts

drawP2:

  ; Zero out offset and offset counter
  lda #$00
  sta PADDLE_OFFSET
  sta $05

  jsr drawP2Loop
  rts

drawP2Loop:
  ; Get latest offset counter
  ldx PADDLE_OFFSET_COUNTER
  inx
  stx PADDLE_OFFSET_COUNTER

  ; Draw current pixel
  ldy PADDLE_OFFSET
  lda #P2_COLOR_VAL
  sta (P2_POSITION_L), Y

  ; Update offset
  tya 
  clc
  adc #ROW_SHIFT_VAL
  sta PADDLE_OFFSET

  ; Update offset counter
  cpx #PADDLE_HEIGHT_VAL
  bne drawP2Loop
  rts

; Memory location $fe contains a new random byte on every instruction.
; Memory location $ff contains the ascii code of the last key pressed.
; 
; Memory locations $200 to $5ff map to the screen pixels. Different values will
; draw different colour pixels. The colours are:
; 
; $0: Black
; $1: White
; $2: Red
; $3: Cyan
; $4: Purple
; $5: Green
; $6: Blue
; $7: Yellow
; $8: Orange
; $9: Brown
; $a: Light red
; $b: Dark grey
; $c: Grey
; $d: Light green
; $e: Light blue
; $f: Light grey
